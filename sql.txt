from (SELECT * FROM order_details WHERE quantity> 10) AS temp_table
group by order_id;

 select * from order_details where quantity> 10
 )

DELIMITER //

DROP FUNCTION IF EXISTS DivideNumbers;

CREATE FUNCTION DivideNumbers(a FLOAT, b FLOAT)
RETURNS FLOAT
DETERMINISTIC
NO SQL
BEGIN
    DECLARE result FLOAT;
    IF b = 0 THEN
        RETURN NULL;
    END IF;
    SET result = a / b;
    RETURN result;
END //

DELIMITER ;

select id, order_id, product_id, quantity, DivideNumbers(quantity, 2) as devided_quantity
from order_details as od
